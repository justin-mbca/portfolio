Cardiovascular Risk Assessment & Clinical Decision Support

Overview

Built a bilingual AI assistant for cardiovascular health assessment that integrates ECG data, cardiac biomarkers (CK-MB, Troponin), and FHIR R4 observations for real-time risk stratification. The system provides clinical decision support for emergency departments and cardiology units.

Challenge

  • Rapid cardiovascular risk assessment in emergency settings
  • Integration of multiple data sources (ECG, labs, vitals)
  • Multi-lingual patient communication
  • FHIR-compliant data exchange
  • Real-time clinical decision support
  • HIPAA-compliant architecture

Solution

  • AI Assistant: LLM-powered conversational interface
  • FHIR Integration: R4 observations and diagnostic reports
  • Biomarker Analysis: Cardiac enzyme interpretation (CK-MB, Troponin)
  • ECG Processing: Medical device data integration
  • Risk Scoring: Automated cardiovascular risk calculation
  • Bilingual Support: English and Spanish interfaces

Code Sample

FHIR Observation Processing (Python)

from fhir.resources.observation import Observation
from datetime import datetime
import gradio as gr

class CardiovascularRiskAssessor:
    def __init__(self):
        self.risk_thresholds = {
            'troponin_i': 0.04,  # ng/mL
            'ck_mb': 5.0,        # ng/mL
            'bnp': 100,          # pg/mL
            'nt_probnp': 125     # pg/mL
        }
    
    def analyze_cardiac_biomarkers(self, fhir_observations: list) -> dict:
        """Analyze cardiac biomarkers from FHIR observations"""
        results = {
            'risk_level': 'LOW',
            'biomarkers': {},
            'recommendations': []
        }
        
        for obs_json in fhir_observations:
            obs = Observation.parse_obj(obs_json)
            
            # Extract biomarker value
            code = obs.code.coding[0].code  # LOINC code
            value = obs.valueQuantity.value
            unit = obs.valueQuantity.unit
            
            # Interpret based on biomarker type
            if code == '10839-9':  # Troponin I
                results['biomarkers']['troponin_i'] = {
                    'value': value,
                    'unit': unit,
                    'elevated': value > self.risk_thresholds['troponin_i']
                }
                
                if value > self.risk_thresholds['troponin_i']:
                    results['risk_level'] = 'HIGH'
                    results['recommendations'].append(
                        "Elevated Troponin I suggests myocardial injury. "
                        "Consider urgent cardiology consultation."
                    )
            
            elif code == '13969-1':  # CK-MB
                results['biomarkers']['ck_mb'] = {
                    'value': value,
                    'unit': unit,
                    'elevated': value > self.risk_thresholds['ck_mb']
                }
                
                if value > self.risk_thresholds['ck_mb']:
                    results['risk_level'] = 'MODERATE'
                    results['recommendations'].append(
                        "Elevated CK-MB. Monitor for acute MI."
                    )
            
            elif code == '30934-4':  # BNP
                results['biomarkers']['bnp'] = {
                    'value': value,
                    'unit': unit,
                    'elevated': value > self.risk_thresholds['bnp']
                }
                
                if value > self.risk_thresholds['bnp']:
                    results['recommendations'].append(
                        "Elevated BNP suggests heart failure. "
                        "Assess for volume overload."
                    )
        
        return results
    
    def calculate_heart_score(self, age: int, risk_factors: dict, 
                             troponin: float, ecg_changes: bool) -> dict:
        """Calculate HEART score for chest pain risk stratification"""
        score = 0
        
        # History (0-2 points)
        if risk_factors.get('typical_angina'):
            score += 2
        elif risk_factors.get('atypical_symptoms'):
            score += 1
        
        # ECG (0-2 points)
        if ecg_changes:
            score += 2
        
        # Age (0-2 points)
        if age >= 65:
            score += 2
        elif age >= 45:
            score += 1
        
        # Risk factors (0-2 points)
        rf_count = sum([
            risk_factors.get('hypertension', False),
            risk_factors.get('diabetes', False),
            risk_factors.get('smoking', False),
            risk_factors.get('family_history', False),
            risk_factors.get('hyperlipidemia', False)
        ])
        
        if rf_count >= 3:
            score += 2
        elif rf_count >= 1:
            score += 1
        
        # Troponin (0-2 points)
        if troponin > 3 * self.risk_thresholds['troponin_i']:
            score += 2
        elif troponin > self.risk_thresholds['troponin_i']:
            score += 1
        
        # Interpret score
        if score <= 3:
            risk = "LOW (2% risk of MACE)"
            disposition = "Consider discharge with outpatient follow-up"
        elif score <= 6:
            risk = "MODERATE (20% risk of MACE)"
            disposition = "Admit for observation and serial troponins"
        else:
            risk = "HIGH (65% risk of MACE)"
            disposition = "Urgent cardiology consultation, consider cath lab"
        
        return {
            'score': score,
            'risk_category': risk,
            'disposition': disposition
        }

# Gradio Interface
def create_ui():
    assessor = CardiovascularRiskAssessor()
    
    with gr.Blocks() as demo:
        gr.Markdown("# CardioInsight AI - Cardiovascular Risk Assessment")
        
        with gr.Tab("Biomarker Analysis"):
            fhir_input = gr.Textbox(
                label="FHIR Observations (JSON)", 
                lines=10
            )
            analyze_btn = gr.Button("Analyze Biomarkers")
            results_output = gr.JSON(label="Risk Assessment")
            
            analyze_btn.click(
                fn=assessor.analyze_cardiac_biomarkers,
                inputs=[fhir_input],
                outputs=[results_output]
            )
        
        with gr.Tab("HEART Score Calculator"):
            age = gr.Slider(18, 100, label="Age")
            troponin = gr.Number(label="Troponin I (ng/mL)")
            ecg_changes = gr.Checkbox(label="Significant ECG Changes")
            
            # Risk factors
            with gr.Group():
                gr.Markdown("**Risk Factors**")
                hypertension = gr.Checkbox(label="Hypertension")
                diabetes = gr.Checkbox(label="Diabetes")
                smoking = gr.Checkbox(label="Current Smoker")
            
            calc_btn = gr.Button("Calculate HEART Score")
            heart_output = gr.JSON(label="HEART Score Result")
    
    return demo

# Launch
if __name__ == "__main__":
    demo = create_ui()
    demo.launch()
