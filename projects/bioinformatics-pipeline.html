Overview

Built a production-ready bioinformatics pipeline for RNA-Seq analysis on AWS, processing 100GB+ genomic datasets with automated ETL, serverless compute, and complete healthcare data standards integration. Achieved 85%+ test coverage with comprehensive CI/CD pipelines.

Challenge

Traditional bioinformatics pipelines lack cloud scalability, healthcare compliance, and modern data engineering practices. Research teams struggle with:

  • Manual data processing workflows
  • Limited compute resources
  • No healthcare data standards integration
  • Lack of automated testing and CI/CD
  • High infrastructure costs

Solution

Migrated analysis workflows to AWS using:

  • AWS Glue for scalable ETL operations
  • EMR for Spark-based analytics
  • Lambda for event-driven automation
  • ECS for containerized workloads
  • FHIR, HL7, OMOP for healthcare interoperability
  • Terraform for infrastructure as code

Data Flow

  1. Ingestion: Raw RNA-Seq data uploaded to S3 Data Lake (raw/ prefix)
  2. Trigger: S3 event triggers Lambda function to initiate processing
  3. ETL: AWS Glue job processes data (filtering, normalization, quality control)
  4. Storage: Processed data saved to S3 (processed/ prefix) in Parquet format
  5. Analysis: EMR cluster performs Spark-based differential expression analysis
  6. Results: Results stored in S3 (results/ prefix) and cataloged in Glue Data Catalog
  7. Integration: Healthcare standards mapping (FHIR, HL7, OMOP) for clinical systems

Code Samples

AWS Glue ETL Job (Python/Spark)

import sys
from awsglue.transforms import *
from awsglue.utils import getResolvedOptions
from pyspark.context import SparkContext
from awsglue.context import GlueContext
from awsglue.job import Job

# Initialize Glue context
args = getResolvedOptions(sys.argv, ['JOB_NAME'])
sc = SparkContext()
glueContext = GlueContext(sc)
spark = glueContext.spark_session
job = Job(glueContext)
job.init(args['JOB_NAME'], args)

# Read from Glue Data Catalog
datasource = glueContext.create_dynamic_frame.from_catalog(
    database="rnaseq_db",
    table_name="raw_counts"
)

# Filter low-count genes (data quality)
filtered = datasource.filter(
    f=lambda x: x["count"] > 10 and x["sample_id"] is not None
)

# Normalize counts (library size normalization)
normalized = filtered.map(
    f=lambda x: {
        **x,
        "normalized_count": x["count"] / x["library_size"] * 1000000
    }
)

# Write to S3 in Parquet format
glueContext.write_dynamic_frame.from_options(
    frame=normalized,
    connection_type="s3",
    connection_options={"path": "s3://pipeline-bucket/processed/"},
    format="parquet",
    format_options={"compression": "snappy"}
)

job.commit()

Terraform Infrastructure (HCL)

# AWS Glue ETL Job
resource "aws_glue_job" "rnaseq_etl" {
  name     = "rnaseq-etl-pipeline"
  role_arn = aws_iam_role.glue_role.arn
  
  command {
    name            = "glueetl"
    script_location = "s3://${var.scripts_bucket}/etl.py"
    python_version  = "3"
  }
  
  default_arguments = {
    "--job-language"        = "python"
    "--enable-metrics"      = "true"
    "--enable-spark-ui"     = "true"
    "--TempDir"             = "s3://${var.temp_bucket}/temp/"
    "--enable-glue-datacatalog" = "true"
  }
  
  max_retries = 1
  timeout     = 60
  
  glue_version       = "3.0"
  number_of_workers  = 10
  worker_type        = "G.1X"
}

# S3 Bucket for Data Lake
resource "aws_s3_bucket" "data_lake" {
  bucket = "bioinformatics-data-lake"
  
  tags = {
    Environment = "production"
    Project     = "RNA-Seq Pipeline"
  }
}

# S3 Lifecycle Policy
resource "aws_s3_bucket_lifecycle_configuration" "data_lake_lifecycle" {
  bucket = aws_s3_bucket.data_lake.id

  rule {
    id     = "archive-old-data"
    status = "Enabled"

    transition {
      days          = 90
      storage_class = "GLACIER"
    }
  }
}

# EMR Cluster
resource "aws_emr_cluster" "spark_cluster" {
  name          = "rnaseq-spark-cluster"
  release_label = "emr-6.10.0"
  applications  = ["Spark", "Hadoop"]

  ec2_attributes {
    subnet_id                         = var.subnet_id
    emr_managed_master_security_group = var.master_sg_id
    emr_managed_slave_security_group  = var.slave_sg_id
    instance_profile                  = aws_iam_instance_profile.emr_profile.arn
  }

  master_instance_group {
    instance_type = "m5.xlarge"
  }

  core_instance_group {
    instance_type  = "m5.xlarge"
    instance_count = 2
  }

  service_role = aws_iam_role.emr_service_role.arn
}

Testing with pytest and moto

import pytest
import boto3
from moto import mock_s3, mock_glue
from etl.glue_job import process_rnaseq_data

@mock_s3
@mock_glue
def test_rnaseq_etl_pipeline():
    # Setup mock AWS services
    s3 = boto3.client('s3', region_name='us-east-1')
    glue = boto3.client('glue', region_name='us-east-1')
    
    # Create test bucket
    bucket_name = 'test-rnaseq-bucket'
    s3.create_bucket(Bucket=bucket_name)
    
    # Upload test data
    test_data = "gene_id,sample_id,count,library_size\nGENE1,SAMPLE1,150,10000000\n"
    s3.put_object(Bucket=bucket_name, Key='raw/test.csv', Body=test_data)
    
    # Run ETL process
    result = process_rnaseq_data(bucket_name, 'raw/test.csv')
    
    # Assertions
    assert result['status'] == 'success'
    assert result['records_processed'] == 1
    assert result['filtered_count'] > 0
    
    # Verify output exists
    response = s3.list_objects_v2(Bucket=bucket_name, Prefix='processed/')
    assert response['KeyCount'] > 0

def test_low_count_filtering():
    """Test that genes with low counts are filtered out"""
    input_data = [
        {"gene_id": "GENE1", "count": 5, "sample_id": "S1"},
        {"gene_id": "GENE2", "count": 150, "sample_id": "S1"},
    ]
    
    filtered = filter_low_counts(input_data, threshold=10)
    
    assert len(filtered) == 1
